cmake_minimum_required(VERSION 3.22)
project(hy300_vpx C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(wrapper "${CMAKE_CURRENT_SOURCE_DIR}/upstream/src/main/jni")
add_library(vpx STATIC IMPORTED)
set_target_properties(vpx PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../.native-video-build/${ANDROID_ABI}/libvpx.a")
add_library(cpufeatures STATIC "${ANDROID_NDK}/sources/android/cpufeatures/cpu-features.c")
target_include_directories(cpufeatures PUBLIC "${ANDROID_NDK}/sources/android/cpufeatures")
add_library(vpxV2JNI SHARED "${wrapper}/vpx_jni.cc")
target_include_directories(vpxV2JNI PRIVATE "${wrapper}/libvpx")
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  target_compile_options(vpxV2JNI PRIVATE -mfpu=neon -marm)
endif()
target_link_libraries(vpxV2JNI PRIVATE vpx cpufeatures android log m z dl)
